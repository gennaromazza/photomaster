4. **🛠️ Controller: transaction, upsert, autorizzazioni**
   1. Nel `selection-controller.ts`, importa `db.transaction` e riscrivi:
      ```ts
      export async function togglePhotoSelection(req, res) {
        const { photoId, sessionId } = req.body;
        try {
          await db.transaction(async (trx) => {
            const exists = await trx
              .select()
              .from(photoSelections)
              .where({ photoId, sessionId });
            if (exists.length) {
              console.log('[DEBUG] Deselezionando', photoId, sessionId);
              await trx
                .delete(photoSelections)
                .where({ photoId, sessionId });
            } else {
              console.log('[DEBUG] Selezionando', photoId, sessionId);
              await trx
                .insert(photoSelections)
                .values({ photoId, sessionId, createdAt: new Date() })
                .onConflict(['photo_id', 'session_id'])
                .ignore();
            }
          });
          res.status(200).json({ success: true });
        } catch (err) {
          console.error('[ERROR] togglePhotoSelection:', err);
          res.status(500).json({ error: err.message });
        }
      }
      ```
   2. **Validazione/autorizzazione** all’inizio del controller:
