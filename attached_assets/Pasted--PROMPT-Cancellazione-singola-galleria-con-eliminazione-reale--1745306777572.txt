1. ROUTE EXPRESS – Aggiungi endpoint DELETE singola galleria
------------------------------------------------------------
TROVARE in routes/gallery.ts:
router.delete("/cleanup", isAuthenticated, cleanupGalleries);

AGGIUNGERE:
router.delete("/galleries/:id", isAuthenticated, async (req, res) => {
  try {
    const { id } = req.params;

    // Recupera la galleria
    const [gallery] = await db.select().from(galleries).where(eq(galleries.id, Number(id)));
    if (!gallery) return res.status(404).json({ error: "Galleria non trovata" });

    // Elimina foto relative e file fisici
    const photosToDelete = await db.select().from(photos).where(eq(photos.galleryId, Number(id)));
    photosToDelete.forEach(photo => {
      if (photo.path && fs.existsSync(photo.path)) fs.unlinkSync(photo.path);
      if (photo.thumbnailPath && fs.existsSync(photo.thumbnailPath)) fs.unlinkSync(photo.thumbnailPath);
      if (photo.mediumPath && fs.existsSync(photo.mediumPath)) fs.unlinkSync(photo.mediumPath);
      if (photo.largePath && fs.existsSync(photo.largePath)) fs.unlinkSync(photo.largePath);
      if (photo.webpPath && fs.existsSync(photo.webpPath)) fs.unlinkSync(photo.webpPath);
    });

    // Elimina da DB
    await db.delete(photos).where(eq(photos.galleryId, Number(id)));
    await db.delete(galleryChapters).where(eq(galleryChapters.galleryId, Number(id)));
    await db.delete(galleries).where(eq(galleries.id, Number(id)));

    return res.json({ success: true });
  } catch (error) {
    console.error("Errore durante la cancellazione:", error);
    return res.status(500).json({ error: "Errore durante la cancellazione" });
  }
});
