async getAllEventClauses(): Promise<EventClause[]> {
  return await db.select().from(eventClauses);
}

async upsertEventClause({ categoryId, content }: { categoryId: number; content: string; }): Promise<EventClause> {
  const [existing] = await db.select().from(eventClauses).where(eq(eventClauses.categoryId, categoryId));
  let result: EventClause;
  if (existing) {
    [result] = await db
      .update(eventClauses)
      .set({ content, updatedAt: new Date() })
      .where(eq(eventClauses.categoryId, categoryId))
      .returning();
  } else {
    [result] = await db
      .insert(eventClauses)
      .values({ categoryId, content })
      .returning();
  }
  return result;
}
